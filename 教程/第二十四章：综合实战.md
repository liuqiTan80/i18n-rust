## 24.4 实现计算

```rust
函数 计算(expr: &表达式) -> 结果<浮点数, 字符串> {
    匹配 expr {
        表达式::数字(n) => 结果::成功(*n),
        表达式::加(左, 右) => {
            让 a = 计算(左)?;
            让 b = 计算(右)?;
            结果::成功(a + b)
        }
        表达式::减(左, 右) => {
            让 a = 计算(左)?;
            让 b = 计算(右)?;
            结果::成功(a - b)
        }
        表达式::乘(左, 右) => {
            让 a = 计算(左)?;
            让 b = 计算(右)?;
            结果::成功(a * b)
        }
        表达式::除(左, 右) => {
            让 a = 计算(左)?;
            让 b = 计算(右)?;
            如果 b == 0.0 {
                结果::错误("除数不能为零".转字符串())
            } 否则 {
                结果::成功(a / b)
            }
        }
    }
}
```

## 24.5 历史记录

用结构体保存历史：

```rust
结构体 历史记录 {
    条目: 向量<历史条目>,
}

结构体 历史条目 {
    表达式: 字符串,
    结果: 浮点数,
}

实现 历史记录 {
    函数 新建() -> 自身 {
        自身 { 条目: 向量::新建() }
    }

    函数 添加(&可变 自我, 表达式: 字符串, 结果: 浮点数) {
        自我.条目.推(历史条目 { 表达式, 结果 });
    }

    函数 显示(&自我) {
        如果 自我.条目.是空() {
            打印行!("暂无历史记录");
            返回;
        }
        打印行!("--- 历史记录 ---");
        对于 (i, 条目) 在 自我.条目.迭代().枚举() {
            打印行!("{}. {} = {}", i + 1, 条目.表达式, 条目.结果);
        }
    }
}
```

## 24.6 主程序

```rust
使用 标准库::io;

函数 主函数() {
    打印行!("=== 命令行计算器 ===");
    打印行!("输入表达式，输入 'quit' 退出");
    打印行!("输入 'history' 查看历史");
    打印行!();

    让 可变 历史 = 历史记录::新建();

    循环 {
        打印!("> ");

        让 可变 输入 = 字符串::新建();
        io::标准输入().读取行(&可变 输入).解包();
        让 输入 = 输入.修剪();

        如果 输入 == "quit" {
            打印行!("再见！");
            跳出;
        }

        如果 输入 == "history" {
            历史.显示();
            继续;
        }

        如果 输入.是空() {
            继续;
        }

        // 简化版：直接解析数字运算
        匹配 简单计算(输入) {
            结果::成功(值) => {
                打印行!("= {}", 值);
                历史.添加(输入.转字符串(), 值);
            }
            结果::错误(原因) => {
                打印行!("错误：{}", 原因);
            }
        }
    }
}
```

## 24.7 简单解析器

为了简化，我们实现一个只支持 `数字 运算符 数字` 的解析器：

```rust
函数 简单计算(输入: &文本) -> 结果<浮点数, 字符串> {
    让 部分: 向量<&文本> = 输入.分割空白().收集();

    如果 部分.长度() != 3 {
        返回 结果::错误("格式错误。请输入：数字 运算符 数字".转字符串());
    }

    让 左 = 部分[0].解析::<浮点数>()
        .或_else(|_| 结果::错误(格式化!("无法解析 '{}'", 部分[0])))?;
    让 运算符 = 部分[1];
    让 右 = 部分[2].解析::<浮点数>()
        .或_else(|_| 结果::错误(格式化!("无法解析 '{}'", 部分[2])))?;

    匹配 运算符 {
        "+" => 结果::成功(左 + 右),
        "-" => 结果::成功(左 - 右),
        "*" => 结果::成功(左 * 右),
        "/" => {
            如果 右 == 0.0 {
                结果::错误("除数不能为零".转字符串())
            } 否则 {
                结果::成功(左 / 右)
            }
        }
        _ => 结果::错误(格式化!("未知运算符：{}", 运算符)),
    }
}
```
