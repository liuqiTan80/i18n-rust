## 24.3 历史记录

> 先做最简单的部分——用结构体保存历史记录。

```rust
结构体 历史记录 {
    条目: 向量<历史条目>,
}

结构体 历史条目 {
    表达式: 字符串,
    数值: 浮点数,
}

实现 历史记录 {
    函数 新建() -> 自身 {
        自身 { 条目: 向量::新建() }
    }

    函数 添加(&可变 自我, 表达式: 字符串, 数值: 浮点数) {
        自我.条目.推(历史条目 { 表达式, 数值 });
    }

    函数 列出(&自我) {
        如果 自我.条目.是否为空() {
            打印行!("暂无历史记录");
            返回;
        }
        打印行!("--- 历史记录 ---");
        让 可变 编号 = 1;
        对于 条目 在 自我.条目.迭代() {
            打印行!("{}. {} = {}", 编号, 条目.表达式, 条目.数值);
            编号 += 1;
        }
    }
}
```

逐行解释：

- 第 1-3 行：`历史记录` 结构体——里面装了一个向量，每个元素是一条记录
- 第 5-8 行：`历史条目` 结构体——每条记录包含"表达式"和"数值"（注意：字段名不能用 `结果`，因为 `结果` 在映射表里对应 `Result`，会被翻译）
- 第 11-13 行：`新建` 方法——创建空的历史记录
- 第 15-17 行：`添加` 方法——往记录里加一条
- 第 19-29 行：`列出` 方法——打印所有记录。用变量 `编号` 从 1 开始数，每打印一条就加 1（这里不用 `.枚举()`，因为 `枚举` 是关键字，对应 `enum`）

## 24.4 计算模块

> 为了简化，我们实现一个只支持 `数字 运算符 数字` 的计算器。

```rust
函数 简单计算(输入: &文本) -> 结果<浮点数, 字符串> {
    // 把输入按空格拆开
    让 部分: 向量<&文本> = 输入.分割空白().收集();

    // 检查格式
    如果 部分.长度() != 3 {
        返回 结果::错误("格式错误。请输入：数字 运算符 数字".转字符串());
    }

    // 解析左边的数字
    让 左 = 部分[0].解析::<浮点数>()
        .或_else(|_| 结果::错误(格式化!("无法解析 '{}'", 部分[0])))?;
    // 读取运算符
    让 运算符 = 部分[1];
    // 解析右边的数字
    让 右 = 部分[2].解析::<浮点数>()
        .或_else(|_| 结果::错误(格式化!("无法解析 '{}'", 部分[2])))?;

    // 根据运算符计算
    匹配 运算符 {
        "+" => 结果::成功(左 + 右),
        "-" => 结果::成功(左 - 右),
        "*" => 结果::成功(左 * 右),
        "/" => {
            如果 右 == 0.0 {
                结果::错误("除数不能为零".转字符串())
            } 否则 {
                结果::成功(左 / 右)
            }
        }
        _ => 结果::错误(格式化!("未知运算符：{}", 运算符)),
    }
}
```

逐行解释：

- 第 1 行：函数返回 `结果` 类型——可能成功（返回数字），也可能失败（返回错误信息）
- 第 3 行：把用户输入按空格拆开——"3 + 5" 变成 `["3", "+", "5"]`
- 第 6-7 行：检查是否恰好 3 部分
- 第 10-11 行：解析左边的数字。`?` 表示如果解析失败就立刻返回错误
- 第 13 行：读取运算符
- 第 15-16 行：解析右边的数字
- 第 19-30 行：用 `匹配` 根据运算符做不同的计算

```
计算流程：

输入："3 + 5"
  ↓ 拆分
["3", "+", "5"]
  ↓ 解析
左 = 3.0, 运算符 = "+", 右 = 5.0
  ↓ 匹配运算符
"+" → 3.0 + 5.0 = 8.0
  ↓ 返回
成功(8.0)
```

## 24.5 主程序

> 把所有部分组合起来——读取用户输入，调用计算，显示结果。

```rust
使用 标准库::io;

函数 主函数() {
    打印行!("=== 命令行计算器 ===");
    打印行!("输入格式：数字 运算符 数字");
    打印行!("输入 'history' 查看历史");
    打印行!("输入 'quit' 退出");
    打印行!();

    让 可变 历史 = 历史记录::新建();

    循环 {
        打印!("> ");

        让 可变 输入 = 字符串::新建();
        io::标准输入().读取行(&可变 输入).解包();
        让 输入 = 输入.裁剪();

        如果 输入 == "quit" {
            打印行!("再见！");
            中断;
        }

        如果 输入 == "history" {
            历史.列出();
            继续;
        }

        如果 输入.是否为空() {
            继续;
        }

        // 调用计算
        匹配 简单计算(输入) {
            结果::成功(数值) => {
                打印行!("= {}", 数值);
                历史.添加(输入.转字符串(), 数值);
            }
            结果::错误(错误信息) => {
                打印行!("错误：{}", 错误信息);
            }
        }
    }
}
```

逐行解释：

- 第 4-8 行：打印欢迎信息
- 第 10 行：创建空的历史记录
- 第 12 行：**`循环`** 无限循环——一直等待用户输入
- 第 13 行：打印提示符 `>`
- 第 15-16 行：读取用户输入
- 第 17 行：去掉首尾空白
- 第 19-22 行：如果输入 "quit" 就退出
- 第 24-27 行：如果输入 "history" 就显示历史（调用 `列出` 方法）
- 第 29-31 行：如果输入为空就跳过
- 第 34-41 行：调用计算函数——成功就显示结果并记录，失败就显示错误
